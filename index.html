export default { async fetch(request, env, ctx) { // CORS başlıkları (Sadece senin GitHub Pages sitene izin verir) const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; // Preflight (OPTIONS) isteklerini yanıtla if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders }); } if (request.method !== "POST") { return new Response("Sadece POST istekleri kabul edilir.", { status: 405, headers: corsHeaders }); } try { const body = await request.json(); const { clientId, clientSecret, boxName, calories, protein, carb, fat } = body; if (!clientId || !clientSecret) { return new Response(JSON.stringify({ error: "Client ID ve Secret eksik!" }), { status: 400, headers: corsHeaders }); } // 1. FatSecret'tan Access Token Al const tokenUrl = "https://oauth.fatsecret.com/connect/token"; const tokenParams = new URLSearchParams({ grant_type: "client_credentials", scope: "basic premier" }); const credentials = btoa(`${clientId}:${clientSecret}`); const tokenRes = await fetch(tokenUrl, { method: "POST", headers: { "Authorization": `Basic ${credentials}`, "Content-Type": "application/x-www-form-urlencoded" }, body: tokenParams }); if (!tokenRes.ok) { const errText = await tokenRes.text(); return new Response(JSON.stringify({ error: "OAuth Hatası: " + errText }), { status: 401, headers: corsHeaders }); } const tokenData = await tokenRes.json(); const accessToken = tokenData.access_token; // 2. FatSecret API'sine Yemeği Ekle const apiUrl = "https://platform.fatsecret.com/rest/server.api"; const apiParams = new URLSearchParams({ method: "food_entry.create", format: "json", food_entry_name: `${boxName} (Pazar Prep)`, calories: calories, protein: protein, carbohydrate: carb, fat: fat, date: "0" // Bugün }); const apiRes = await fetch(apiUrl, { method: "POST", headers: { "Authorization": `Bearer ${accessToken}`, "Content-Type": "application/x-www-form-urlencoded" }, body: apiParams }); const apiData = await apiRes.json(); return new Response(JSON.stringify(apiData), { headers: { ...corsHeaders, "Content-Type": "application/json" } }); } catch (err) { return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: corsHeaders }); } } };
